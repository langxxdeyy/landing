<?php
header("Content-Type: application/xml; charset=utf-8");
$brandsFile = __DIR__ . "/cola.txt";
$base = "https://digimarconvienna.com/about/";
$lines = @file($brandsFile, FILE_IGNORE_NEW_LINES|FILE_SKIP_EMPTY_LINES);
if ($lines === false) $lines = [];
echo '<?xml version="1.0" encoding="UTF-8"?>' . "\n";
?>
<urlset xmlns="http://www.sitemaps.org/schemas/sitemap/0.9">
<?php foreach ($lines as $brand): $b = trim($brand); if ($b === "") continue; ?>
  <url>
    <loc><?php echo htmlspecialchars($base . "?ID=" . urlencode(strtoupper($b)), ENT_XML1, "UTF-8"); ?></loc>
    <changefreq>weekly</changefreq>
    <priority>0.7</priority>
  </url>
<?php endforeach; ?>
</urlset>
